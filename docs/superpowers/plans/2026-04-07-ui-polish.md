function getBestPR(exerciseData) {
  let best = null;
  Object.entries(exerciseData || {}).forEach(([reps, entries]) => {
    (entries || []).forEach((entry) => {
      const w = parseFloat(entry.weight);
      if (!isNaN(w) && (best === null || w > best.weight || (w === best.weight && parseInt(reps) < best.reps))) {
        best = { weight: w, reps: parseInt(reps) };
      }
    });
  });
  return best;
}
```

function Sparkline({ data, width = 72, height = 36, colour = '#4A90D9' }) {
  if (data.length < 2) return null;
  const weights = data.map((d) => d.weight);
  const min = Math.min(...weights);
  const max = Math.max(...weights);
  const range = max - min || 1;

  const points = data
    .map((d) => {
      const x = (d.day / 27) * width;
      const y = height - 4 - ((d.weight - min) / range) * (height - 8);
      return `${x.toFixed(1)},${y.toFixed(1)}`;
    })
    .join(' ');

  const last = data[data.length - 1];
  const lx = ((last.day / 27) * width).toFixed(1);
  const ly = (height - 4 - ((last.weight - min) / range) * (height - 8)).toFixed(1);

  return (
    <svg width={width} height={height} style={{ overflow: 'visible', flexShrink: 0, display: 'block' }}>
      <polyline points={points} fill="none" stroke={colour} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
      <circle cx={lx} cy={ly} r="2.5" fill={colour} />
    </svg>
  );
}
```

// Wave timing (waves 3-4 are 1.7× faster per spec)
const WAVE_TIMING = {
  1: { duration: 0.85, delay: 0.05 },
  2: { duration: 0.85, delay: 0.20 },
  3: { duration: 0.50, delay: 0.30 },
  4: { duration: 0.50, delay: 0.38 },
};

const EASING = 'cubic-bezier(.2,.8,.35,1)';

// Replicates Dashboard's getMuscleColor logic — returns null if untrained
function getMuscleColour(muscleKey, weeklyVolume, volumeTargets) {
  const sets = weeklyVolume[muscleKey] || 0;
  if (sets === 0) return null;
  const target = volumeTargets[muscleKey] || 10;
  const ratio = sets / target;
  if (ratio < 0.5) return '#4a5a3a';
  if (ratio < 1.0) return '#c8b400';
  if (ratio < 1.2) return '#5BBD72';
  return '#F0A500';
}

// CSS keyframes injected once
const KEYFRAMES = `
  @keyframes __mao_fromLeft {
    0%   { opacity:0; transform: translateX(-120%) scale(0.75); }
    65%  { opacity:1; transform: translateX(4%)    scale(1.05); }
    82%  { transform: translateX(-2%) scale(0.98); }
    100% { opacity:1; transform: translateX(0)     scale(1); }
  }
  @keyframes __mao_fromRight {
    0%   { opacity:0; transform: translateX(120%)  scale(0.75); }
    65%  { opacity:1; transform: translateX(-4%)   scale(1.05); }
    82%  { transform: translateX(2%)  scale(0.98); }
    100% { opacity:1; transform: translateX(0)     scale(1); }
  }
  @keyframes __mao_flash {
    0%   { filter: brightness(1); }
    25%  { filter: brightness(4.5); }
    100% { filter: brightness(1); }
  }
`;

let keyframesInjected = false;
function ensureKeyframes() {
  if (keyframesInjected) return;
  const style = document.createElement('style');
  style.textContent = KEYFRAMES;
  document.head.appendChild(style);
  keyframesInjected = true;
}

export default function MuscleAnimationOverlay({ weeklyVolume, volumeTargets, phase }) {
  // phase: 'idle' | 'expanding' | 'animating' | 'settling' | 'contracting'

  useEffect(() => { ensureKeyframes(); }, []);

  // Fire haptics when animation begins
  useEffect(() => {
    if (phase !== 'animating' || !navigator.vibrate) return;
    const timers = Object.values(WAVE_TIMING).map(({ delay, duration }) => {
      const ms = (delay + duration * 0.65) * 1000;
      return setTimeout(() => navigator.vibrate(25), ms);
    });
    return () => timers.forEach(clearTimeout);
  }, [phase]);

  if (phase === 'idle') return null;

  // Build list of muscle shapes to render
  const shapes = [];
  Object.entries(ANIM_CONFIG).forEach(([muscleKey, config]) => {
    const colour = getMuscleColour(muscleKey, weeklyVolume, volumeTargets);
    if (!colour) return;
    const timing = WAVE_TIMING[config.wave];
    config.shapes.forEach((shape, i) => {
      const fromRight = config.from === 'sides' ? i === 1 : config.from === 'right';
      shapes.push({ ...shape, colour, muscleKey, fromRight, timing, key: `${muscleKey}-${i}` });
    });
  });

  const isExpanded = phase === 'expanding' || phase === 'animating' || phase === 'settling';
  const overlayAlpha = isExpanded ? 0.95 : 0;
  const stageSize = isExpanded ? 'min(72vw, 340px)' : '140px';
  const stageBottom = isExpanded ? '50%' : '80px';
  const stageTransform = isExpanded ? 'translateX(-50%) translateY(50%)' : 'translateX(-50%)';
  const stageDuration = phase === 'contracting' ? '0.55s' : '0.45s';

  return (
    <>
      {/* Dark overlay */}
      <div style={{
        position: 'fixed', inset: 0, zIndex: 200,
        background: `rgba(0,0,0,${overlayAlpha})`,
        transition: 'background 0.35s ease',
        pointerEvents: isExpanded ? 'all' : 'none',
      }} />

      {/* Muscle stage */}
      <div style={{
        position: 'fixed', zIndex: 210,
        width: stageSize, height: stageSize,
        bottom: stageBottom, left: '50%',
        transform: stageTransform,
        transition: `width ${stageDuration} cubic-bezier(.4,0,.2,1), height ${stageDuration} cubic-bezier(.4,0,.2,1), bottom ${stageDuration} cubic-bezier(.4,0,.2,1), transform ${stageDuration} cubic-bezier(.4,0,.2,1)`,
        overflow: 'hidden',
      }}>
        <svg viewBox="0 0 100 200" width="100%" height="100%" style={{ overflow: 'visible' }}>
          {/* Head */}
          <ellipse cx="50" cy="13" rx="16" ry="12" fill="#1e1e1e" stroke="#2a2a2a" strokeWidth="0.8" />

          {/* Animated trained muscles (phase: animating) */}
          {phase === 'animating' && shapes.map((s) => {
            const dir = s.fromRight ? '__mao_fromRight' : '__mao_fromLeft';
            const flashDelay = s.timing.delay + s.timing.duration * 0.65;
            return (
              <rect
                key={s.key}
                x={s.x} y={s.y} width={s.w} height={s.h} rx={s.rx}
                fill={s.colour}
                style={{
                  animation: `${dir} ${s.timing.duration}s ${s.timing.delay}s ${EASING} both, __mao_flash 0.22s ${flashDelay}s ease-out both`,
                }}
              />
            );
          })}

          {/* Static trained muscles after animation (settling, contracting) */}
          {(phase === 'settling' || phase === 'contracting') && shapes.map((s) => (
            <rect
              key={s.key}
              x={s.x} y={s.y} width={s.w} height={s.h} rx={s.rx}
              fill={s.colour}
            />
          ))}
        </svg>
      </div>
    </>
  );
}

useEffect(() => {

  let cancelled = false;

  const schedule = (fn, ms) => setTimeout(() => { if (!cancelled) fn(); }, ms);

  setMomentPhase('expanding');
  const t1 = schedule(() => setMomentPhase('animating'),   500);
  const t2 = schedule(() => setMomentPhase('settling'),    1500);
  const t3 = schedule(() => setMomentPhase('contracting'), 2050);
  const t4 = schedule(() => setMomentPhase('idle'),        2650);

  return () => {
    cancelled = true;
    clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4);

**Note:** The `eslint-disable` comment suppresses the exhaustive-deps warning. The effect is intentionally mount-only — the moment plays once per Dashboard mount. `weeklyVolume` is checked at the time the effect runs (it will be `{}` on first mount if data hasn't loaded yet, which means the moment won't play on a fresh install with no data, which is correct behaviour per the spec).

To handle the case where `weeklyVolume` is loaded asynchronously (after mount), revise the approach: delay the moment start until after the data-loading effect has run, by using a `loaded` flag:

Replace the above `useEffect` with this two-part pattern:

```jsx
const [dataLoaded, setDataLoaded] = useState(false);

// In the existing data-loading useEffect, at the very end, add:
setDataLoaded(true);

// New effect — runs once after data is loaded
useEffect(() => {
  if (!dataLoaded) return;
  const hasTrainedMuscles = Object.values(weeklyVolume).some((v) => v > 0);
  if (!hasTrainedMuscles) return;

  let cancelled = false;
  const schedule = (fn, ms) => setTimeout(() => { if (!cancelled) fn(); }, ms);

  setMomentPhase('expanding');
  const t1 = schedule(() => setMomentPhase('animating'),   500);
  const t2 = schedule(() => setMomentPhase('settling'),    1500);
  const t3 = schedule(() => setMomentPhase('contracting'), 2050);
  const t4 = schedule(() => setMomentPhase('idle'),        2650);

  return () => {
    cancelled = true;
    [t1, t2, t3, t4].forEach(clearTimeout);
  };
}, [dataLoaded]); // eslint-disable-line react-hooks/exhaustive-deps
```

Don't forget to add `const [dataLoaded, setDataLoaded] = useState(false);` to the state declarations, and `setDataLoaded(true);` at the end of the existing data-loading `useEffect`.
